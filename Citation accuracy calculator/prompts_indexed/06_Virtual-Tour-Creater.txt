  52 | def calculate_view_constraints(image):
  53 |     """
  54 |     Calculates view constraints based on non-black pixel boundaries.
  55 |     Uses more robust contour detection and handles potential edge cases.
  56 |     """
  57 |     if image is None:
  58 |         return {'minYaw': 0, 'maxYaw': 2 * np.pi, 'minPitch': -np.pi / 2, 'maxPitch': np.pi / 2}
  59 | 
  60 |     gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)
  61 |     _, thresh = cv2.threshold(gray, 10, 255, cv2.THRESH_BINARY)
  62 |     thresh = cv2.erode(thresh, None, iterations=2)  # Reduce noise
  63 |     thresh = cv2.dilate(thresh, None, iterations=2) # Fill gaps
  64 | 
  65 |     contours, _ = cv2.findContours(thresh, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
  66 | 
  67 |     if not contours:
  68 |         return {'minYaw': 0, 'maxYaw': 2 * np.pi, 'minPitch': -np.pi / 2, 'maxPitch': np.pi / 2}
  69 | 
  70 |     c = max(contours, key=cv2.contourArea)
  71 |     mask = np.zeros_like(gray)
  72 |     cv2.drawContours(mask, [c], -1, 255, -1)
  73 |     y_coords, x_coords = np.where(mask > 0)
  74 | 
  75 |     if not x_coords.size or not y_coords.size:
  76 |         return {'minYaw': 0, 'maxYaw': 2 * np.pi, 'minPitch': -np.pi / 2, 'maxPitch': np.pi / 2}
  77 | 
  78 |     min_x, max_x = np.min(x_coords), np.max(x_coords)
  79 |     min_y, max_y = np.min(y_coords), np.max(y_coords)
  80 | 
  81 |     height, width = image.shape[:2]
  82 |     center_x = width / 2
  83 |     center_y = height / 2
  84 | 
  85 |     # Convert pixel coordinates to angles (simplified)
  86 |     # These calculations are approximations and might need fine-tuning
  87 |     # based on the specific panorama projection and desired behavior.
  88 |     # For a perfect equirectangular projection, more complex formulas are needed.
  89 |     # Here, we're doing a linear mapping based on bounding box.
  90 |     left_angle = np.arctan2(min_x - center_x, center_y)
  91 |     right_angle = np.arctan2(max_x - center_x, center_y)
  92 |     top_angle = np.arctan2(center_x, min_y - center_y)
  93 |     bottom_angle = np.arctan2(center_x, max_y - center_y)

   1 | import cv2
   2 | import numpy as np
   3 | 
   4 | def stitch_images(image_paths, output_path):
   5 |     """
   6 |     Stitches images together to create a panorama and attempts to remove black areas.
   7 | 
   8 |     Args:
   9 |         image_paths (list): List of paths to the images to stitch.
  10 |         output_path (str): Path to save the stitched panorama.
  11 | 
  12 |     Returns:
  13 |         tuple: (bool, numpy.ndarray)
  14 |             - The first element is a boolean indicating success (True) or failure (False).
  15 |             - The second element is the stitched image as a numpy.ndarray if successful, None otherwise.
  16 |     """
  17 |     images = [cv2.imread(path) for path in image_paths if path]
  18 | 
  19 |     if len(images) < 2:
  20 |         return False, None  # Need at least 2 images to stitch
  21 | 
  22 |     stitcher = cv2.Stitcher_create()
  23 |     status, stitched = stitcher.stitch(images)
  24 | 
  25 |     if status == cv2.Stitcher_OK:
  26 |         stitched_image = stitched
  27 |         # Attempt to remove black borders
  28 |         stitched_image = remove_black_borders(stitched_image)
  29 |         cv2.imwrite(output_path, stitched_image)
  30 |         return True, stitched_image
  31 |     else:
  32 |         print("Stitching failed with status code:", status)
  33 |         return False, None
  34 | 
  35 | def remove_black_borders(image):
  36 |     """
  37 |     Removes black borders from a stitched image.
  38 | 
  39 |     Args:
  40 |         image (numpy.ndarray): The stitched image.
  41 | 
  42 |     Returns:
  43 |         numpy.ndarray: The image with black borders removed, or the original image if no borders were found.
  44 |     """
  45 |     # 1. Convert to grayscale
  46 |     gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)
  47 | 
  48 |     # 2. Find the non-black regions
  49 |     _, thresh = cv2.threshold(gray, 1, 255, cv2.THRESH_BINARY)
  50 |     contours, _ = cv2.findContours(thresh, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
  51 | 
  52 |     if not contours:
  53 |         return image  # No non-black regions found
  54 | 
  55 |     # 3. Get the bounding rectangle of the non-black region
  56 |     x, y, w, h = cv2.boundingRect(contours[0])
  57 | 
  58 |     # 4. Crop the image to the bounding rectangle
  59 |     cropped_image = image[y:y+h, x:x+w]
  60 |     return cropped_image

// UploadComponent.js
import React, { useState } from 'react';
import { uploadImageToSupabase } from './uploadToSupabase';

export default function UploadComponent() {
  const [file, setFile] = useState(null);
  const [url, setUrl] = useState('');
  const [status, setStatus] = useState('');

  const handleUpload = async () => {
    setStatus('Uploading...');
    const result = await uploadImageToSupabase(file, 'yasha@example.com');
    if (result.error) {
      setStatus('Error: ' + result.error);
    } else {
      setUrl(result.url);
      setStatus('Upload successful!');
    }
  };

  return (
    <div>
      <input
        type="file"
        accept="image/*"
        onChange={(e) => setFile(e.target.files[0])}
      />
      <button onClick={handleUpload}>Upload</button>
      <p>{status}</p>
      {url && <img src={url} alt="Uploaded" style={{ maxWidth: '300px' }} />}
    </div>
  );
}

import React, { useState } from "react";
import { BrowserRouter as Router, Routes, Route, Navigate } from "react-router-dom";
import LoginPage from "./LoginPage";  // Import your LoginPage component
import StegoApp from "./StegoApp";
import UploadComponent from './UploadComponent';

function App() {
  const [isAuthenticated, setIsAuthenticated] = useState(false);
  
  // Function to handle successful login
  const handleLoginSuccess = () => {
    setIsAuthenticated(true);
  };

  return (
    <Router>
      <Routes>
        {/* Login route - show login page if not authenticated */}
        <Route 
          path="/login" 
          element={
            isAuthenticated ? 
            <Navigate to="/app" /> : 
            <LoginPage onLoginSuccess={handleLoginSuccess} />
          } 
        />
        
        {/* Main app route - requires authentication */}
        <Route 
          path="/app" 
          element={
            isAuthenticated ? 
            <StegoApp /> : 
            <Navigate to="/login" />
          } 
        />
        
        {/* Upload component route */}
        <Route 
          path="/upload" 
          element={
            isAuthenticated ? 
            <UploadComponent /> : 
            <Navigate to="/login" />
          } 
        />
        
        {/* Default route - redirect to login if not authenticated, app if authenticated */}
        <Route 
          path="*" 
          element={
            isAuthenticated ? 
            <Navigate to="/app" /> : 
            <Navigate to="/login" />
          } 
        />
      </Routes>
    </Router>
  );
}

export default App;

// Simplified upload function for debugging
export async function uploadImageToSupabase(file, encryptionKey = "") {
  if (!file) return { error: 'No file provided' };

  try {
    const { data: authData } = await supabase.auth.getSession();
    if (!authData.session) return { error: 'Not authenticated' };

    const email = authData.session.user.email;
    const filePath = `stego/${Date.now()}-${file.name}`;

    const { data, error } = await supabase
      .storage
      .from('stego-images')
      .upload(filePath, file, {
        cacheControl: '3600',
        upsert: false,
      });

    if (error) return { error: error.message };

    // Get public URL
    const { data: urlData } = supabase.storage.from('stego-images').getPublicUrl(filePath);
    const publicUrl = urlData.publicUrl;

    // Insert metadata into uploads table
    const { error: insertError } = await supabase
      .from('uploads')
      .insert([{
        user_email: email,
        file_name: file.name,
        public_url: publicUrl,
        encryption_key: encryptionKey
      }]);

    if (insertError) return { error: insertError.message };

    return { url: publicUrl, success: true };
  } catch (err) {
    return { error: err.message };
  }
}
